cmake_minimum_required(VERSION 3.12)
project(MIR2Neural VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(STRICT "Enable strict compiler diagnostics" ON)
option(MIR2_NEURAL_SANITIZE "Enable sanitizers for native probes and tests" OFF)
option(MIR2_NEURAL_PLUGIN "Build the optional SuperCollider extension" OFF)

add_library(mir2_neural INTERFACE)
target_include_directories(mir2_neural INTERFACE include)
if(STRICT)
    if(MSVC)
        target_compile_options(mir2_neural INTERFACE /W4)
    else()
        target_compile_options(mir2_neural INTERFACE -Wall -Wextra -Wpedantic -Wconversion)
    endif()
endif()

if(MIR2_NEURAL_PLUGIN)
    option(SCSYNTH "Build scsynth plugin" ON)
    option(SUPERNOVA "Build supernova plugin" OFF)
    if(NOT SC_PATH)
        message(FATAL_ERROR "SC_PATH must point to a matching SuperCollider source tree")
    endif()
    get_filename_component(sc_source "${SC_PATH}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    list(APPEND CMAKE_MODULE_PATH "${sc_source}/tools/cmake_gen")
    include(SuperColliderServerPlugin)
    sc_do_initial_compiler_config()
    sc_check_sc_path("${sc_source}")
    sc_add_server_plugin("MIR2Neural" "MIR2Neural" "plugins/MIR2Neural.cpp"
        "Classes/PitchPESTO.sc" "HelpSource/Classes/PitchPESTO.schelp")
    foreach(target MIR2Neural_scsynth MIR2Neural_supernova)
        if(TARGET ${target})
            target_link_libraries(${target} PRIVATE mir2_neural)
            if(MINGW)
                target_link_options(${target} PRIVATE -static-libgcc -static-libstdc++)
            endif()
            target_include_directories(${target} SYSTEM PRIVATE
                "${sc_source}/include/plugin_interface" "${sc_source}/include/common"
                "${sc_source}/common")
            if(NOT MSVC)
                target_compile_options(${target} PRIVATE -Wno-unused-parameter)
            endif()
        endif()
    endforeach()
    if(TARGET MIR2Neural_scsynth)
        set_target_properties(MIR2Neural_scsynth PROPERTIES OUTPUT_NAME "MIR2Neural")
    endif()
    if(TARGET MIR2Neural_supernova)
        set_target_properties(MIR2Neural_supernova PROPERTIES OUTPUT_NAME "MIR2Neural_supernova")
    endif()
    install(FILES README.md THIRD_PARTY_NOTICES.md ../LICENSE DESTINATION MIR2Neural)
    install(DIRECTORY licenses/ DESTINATION MIR2Neural/licenses)
    install(FILES tools/export_pesto.py tools/requirements-reference.txt DESTINATION MIR2Neural/tools)
endif()
if(MIR2_NEURAL_SANITIZE)
    target_compile_options(mir2_neural INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(mir2_neural INTERFACE -fsanitize=address,undefined)
endif()
add_executable(pesto_probe tools/pesto_probe.cpp)
target_link_libraries(pesto_probe PRIVATE mir2_neural)
add_executable(pesto_stream_probe tools/pesto_stream_probe.cpp)
target_link_libraries(pesto_stream_probe PRIVATE mir2_neural)
include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    file(GLOB tests CONFIGURE_DEPENDS tests/*.cpp)
    foreach(source IN LISTS tests)
        get_filename_component(name "${source}" NAME_WE)
        add_executable(${name} "${source}")
        target_link_libraries(${name} PRIVATE mir2_neural)
        if(MSVC)
            target_compile_options(${name} PRIVATE /UNDEBUG)
        else()
            target_compile_options(${name} PRIVATE -UNDEBUG)
        endif()
        add_test(NAME ${name} COMMAND ${name})
    endforeach()
endif()
